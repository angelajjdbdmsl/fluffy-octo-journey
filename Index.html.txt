<script>
const canciones = [
  { titulo:"SoundHelix 1 - Chill", artista:"TONY Beats", color:"#ff2d55", emoji:"🔥", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-1.mp3" },
  { titulo:"SoundHelix 2 - Party", artista:"TONY Beats", color:"#ff9500", emoji:"🎉", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-2.mp3" },
  { titulo:"SoundHelix 3 - Noche", artista:"TONY Beats", color:"#8b5cf6", emoji:"🌙", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-3.mp3" },
  { titulo:"SoundHelix 4 - Gym", artista:"TONY Beats", color:"#22c55e", emoji:"💪", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-4.mp3" },
  { titulo:"SoundHelix 5 - Relax", artista:"TONY Beats", color:"#3b82f6", emoji:"😎", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-5.mp3" },
  { titulo:"SoundHelix 6 - Flow", artista:"TONY Beats", color:"#ec4899", emoji:"✨", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-6.mp3" },
  { titulo:"SoundHelix 7 - Drive", artista:"TONY Beats", color:"#f43f5e", emoji:"🚗", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-7.mp3" },
  { titulo:"SoundHelix 8 - Fuego", artista:"TONY Beats", color:"#eab308", emoji:"🔥", url:"https://www.soundhelix.com/examples/mp3/SoundHelix-Song-8.mp3" },
];

let indiceActual = -1;
let listaFiltrada = [...canciones];
const audio = document.getElementById('audio');

function renderizar(){
  const cont = document.getElementById('lista');
  cont.innerHTML = '';
  listaFiltrada.forEach((c, i)=>{
    const idx = canciones.indexOf(c);
    const div = document.createElement('div');
    div.className = 'card' + (idx===indiceActual?' active':'');
    div.innerHTML = `<div class="cover" style="background:${c.color}">${c.emoji}</div><div class="card-info"><b>${c.titulo}</b><span>${c.artista}</span></div><button class="play-btn">${idx===indiceActual && !audio.paused?'⏸️':'▶️'}</button>`;
    div.onclick = ()=> reproducir(idx);
    cont.appendChild(div);
  });
  if(listaFiltrada.length===0) cont.innerHTML = '<p style="text-align:center;color:#888;padding:20px">No encontré esa rola 😢<br>Sube tu propio MP3 con el botón de abajo</p>';
}

function reproducir(idx){
  indiceActual = idx;
  const c = canciones[idx];
  audio.src = c.url;
  audio.play();
  document.getElementById('tituloActual').innerText = c.titulo;
  document.getElementById('artistaActual').innerText = c.artista;
  document.getElementById('coverActual').innerText = c.emoji;
  document.getElementById('coverActual').style.background = c.color;
  document.getElementById('btnPlay').innerText = '⏸️';
  renderizar();
}

function togglePlay(){
  if(indiceActual===-1){ reproducir(0); return; }
  if(audio.paused){ audio.play(); document.getElementById('btnPlay').innerText='⏸️'; }
  else{ audio.pause(); document.getElementById('btnPlay').innerText='▶️'; }
  renderizar();
}

function siguiente(){
  if(indiceActual===-1) return;
  reproducir((indiceActual+1)%canciones.length);
}
function anterior(){
  if(indiceActual===-1) return;
  reproducir((indiceActual-1+canciones.length)%canciones.length);
}

function buscar(){
  const q = document.getElementById('buscador').value.toLowerCase();
  listaFiltrada = canciones.filter(c=> (c.titulo+' '+c.artista).toLowerCase().includes(q));
  renderizar();
}

function cambiarVolumen(v){ audio.volume = v; }
function agregarCancion(){ document.getElementById('inputFile').click(); }
function subirMusica(e){
  [...e.target.files].forEach(f=>{
    const url = URL.createObjectURL(f);
    canciones.push({ titulo:f.name.replace('.mp3','').replace('.m4a',''), artista:"Mi Música", color:"#ff2d55", emoji:"🎵", url });
  });
  listaFiltrada = [...canciones];
  renderizar();
  alert('✅ Agregaste ' + e.target.files.length + ' canción(es)');
}

audio.addEventListener('timeupdate', ()=>{
  if(!audio.duration) return;
  const p = (audio.currentTime/audio.duration)*100;
  document.getElementById('barra').style.width = p+'%';
  document.getElementById('tiempoActual').innerText = formato(audio.currentTime);
  document.getElementById('tiempoTotal').innerText = formato(audio.duration);
});
audio.addEventListener('ended', ()=> siguiente());
function formato(s){ let m=Math.floor(s/60), ss=Math.floor(s%60); return m+':'+(ss<10?'0'+ss:ss); }
function buscarTiempo(e){
  const rect = e.currentTarget.getBoundingClientRect();
  const p = (e.clientX-rect.left)/rect.width;
  if(audio.duration) audio.currentTime = p*audio.duration;
}

renderizar();
